    61→  const failed = job.status === "failed" || job.status === "failed_permanent";
    62→  const done = job.status === "done";
    63→  const [liveOpen, setLiveOpen] = useState(false);
    64→  const [editorOpen, setEditorOpen] = useState(false);

    67→
    68→  const timings = job.stage_timings || {};
    69→  const progress = job.progress ?? 0;

    71→
    72→  // ticking clock for elapsed / ETA readouts (presentational only)
    73→  useEffect(() => {
    74→    if (!active) return;
    75→    const id = setInterval(() => setNow(Date.now()), 1000);
    76→    return () => clearInterval(id);
    77→  }, [active]);
    78→
    79→  // cancel needs a second tap within 3s — protects against accidental taps
    80→  useEffect(() => {
    81→    if (!confirmCancel) return;
    82→    const id = setTimeout(() => setConfirmCancel(false), 3000);
    83→    return () => clearTimeout(id);
    84→  }, [confirmCancel]);
    85→
    86→  const startMs = job.start_time > 1e12 ? job.start_time : job.start_time * 1000;
    87→  const elapsedSec = active && startMs > 0 ? Math.max(0, (now - startMs) / 1000) : 0;
    88→  const etaSec = active && progress > 0.08 && progress < 0.95 && elapsedSec > 3
    89→    ? (elapsedSec * (1 - progress)) / progress
    90→    : null;
    91→
    92→  const cancelClick = () => {
    93→    if (!confirmCancel) {
    94→      setConfirmCancel(true);
    95→      return;
    96→    }
    97→    setConfirmCancel(false);

   100→
   101→  // pipeline line fill: reaches the current stage node (1.0 when finished)
   102→  const stageIdx = STAGES.indexOf(job.stage as typeof STAGES[number]);
   103→  const lineFrac = done ? 1 : Math.max(0, Math.min(3, stageIdx)) / 3;
   104→
   105→  // skeleton placeholders while clips render (n = expected clips)
   106→  const expected = settings.clip.maxClips === "auto" ? 3 : parseInt(settings.clip.maxClips, 10) || 3;
   107→  const rendering = active && job.status === "rendering";
   108→  const skeletons = rendering ? Math.max(0, expected - (job.clips?.length ?? 0)) : 0;
   109→
   110→  return (
   111→    <div className="space-y-4 pb-4">
   112→      {/* top bar: back + title */}
   113→      <div className="flex items-center gap-3 pt-1">
   114→        <button
   115→          type="button"
   116→          onClick={onBack}
   117→          aria-label="Back"

   121→        </button>
   122→        <div className="min-w-0">
   123→          <h2 className="truncate text-[15px] font-bold leading-tight tracking-tight text-white">
   124→            {active ? "Generating clips" : failed ? "Job failed" : job.status === "canceled" ? "Canceled" : "Clips ready"}
   125→          </h2>
   126→          <p className="truncate text-[11px] text-zinc-500">
   127→            {job.video_title || job.source_url || "Uploaded video"}
   128→          </p>
   129→        </div>
   130→      </div>
   131→
   132→      {/* ACTIVE — pipeline hero */}
   133→      {active && (
   134→        <section aria-label="Job status" className="glass ac-rise overflow-hidden rounded-3xl">
   135→          <div className="p-5">
   136→            {/* pipeline visualization */}
   137→            <div
   138→              className="relative mx-auto max-w-[310px]"
   139→              role="progressbar"
   140→              aria-valuenow={Math.round(progress * 100)}
   141→              aria-valuemin={0}
   142→              aria-valuemax={100}
   143→              aria-label={STAGE_LABELS[job.stage] || job.stage}
   144→            >
   145→              <div className="relative flex items-start justify-between">
   146→                {/* base track + gradient fill between node centers */}
   147→                <span aria-hidden="true" className="absolute left-[12.5%] right-[12.5%] top-[22px] h-[3px] rounded-full bg-white/10" />
   148→                <span
   149→                  aria-hidden="true"
   150→                  className="absolute left-[12.5%] top-[22px] h-[3px] rounded-full bg-gradient-to-r from-violet-500 to-fuchsia-500 transition-all duration-700 ease-out"
   151→                  style={{ width: `${75 * lineFrac}%` }}
   152→                />
   153→                {STAGES.map((s, i) => {
   154→                  const stageDone = timings[s] !== undefined;
   155→                  // when queued (stage not yet known) the first node pulses to show liveness
   156→                  const current = job.stage === s || (job.status === "queued" && i === 0);
   157→                  const Icon = STAGE_ICON[s];
   158→                  return (
   159→                    <div key={s} className="relative z-10 flex w-[56px] flex-col items-center gap-1.5">
   160→                      <span

   164→                            : current

   176→                        {STAGE_SHORT[s]}

   311→            <span
   312→              className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-2xl ring-1 ring-inset ${

   320→              <h2 className="text-[15px] font-bold text-white">
   321→                {failed ? "Couldn't finish this job" : "Job canceled"}
   322→              </h2>

   329→          </div>

   408→          type="button"
   409→          onClick={() => setEditorOpen(true)}
   410→          aria-label="Edit transcript"
   411→          className="ac-focus glass flex h-11 w-full items-center justify-center gap-2 rounded-2xl text-[12px] font-semibold text-zinc-300 transition-all duration-200 active:scale-[0.98] hover:text-violet-200"

   464→function ClipCard({ job, clip }: { job: Job; clip: Clip }) {
   465→  const url = clipFileUrl(job.id, clip);
   466→  // scores may arrive on a 0–10 or 0–100 scale depending on scorer version
   467→  const norm = clip.score <= 10 ? clip.score / 10 : clip.score / 100;

   547→/** Engine-rendered single frame preview with time slider. */
   548→function LivePreviewDialog({
   549→  job, settings, open, onOpenChange,
   550→}: { job: Job; settings: ClipSettings; open: boolean; onOpenChange: (v: boolean) => void }) {
   551→  const [t, setT] = useState(Math.min(5, Math.max(1, (job.duration || 10) / 2)));
   552→  const [img, setImg] = useState<string | null>(null);
   553→  const [loading, setLoading] = useState(false);
   554→  const debounce = useRef<ReturnType<typeof setTimeout> | null>(null);
   555→
   556→  const renderPreview = useCallback(
   557→    async (time: number) => {
   558→      setLoading(true);
   559→      try {
   560→        const res = await fetch(
   561→          `/engine/api/jobs/${job.id}/preview?XTransformPort=${ENGINE_PORT}`,
   562→          {
   563→            method: "POST",
   564→            headers: { "Content-Type": "application/json" },
   565→            body: JSON.stringify({ t: time, settings }),
   566→          }
   567→        );
   568→        if (!res.ok) throw new Error("preview failed");
   569→        const blob = await res.blob();
   570→        setImg((old) => {
   571→          if (old) URL.revokeObjectURL(old);
   572→          return URL.createObjectURL(blob);
   573→        });
   574→      } catch {
   575→        /* preview is best-effort */
   576→      } finally {
   577→        setLoading(false);
   578→      }
   579→    },
   580→    [job.id, settings]
   581→  );
   582→
   583→  useEffect(() => {
   584→    if (!open) return;
   585→    if (debounce.current) clearTimeout(debounce.current);
   586→    debounce.current = setTimeout(() => renderPreview(t), 400);
   587→    return () => {
   588→      if (debounce.current) clearTimeout(debounce.current);
   589→    };
   590→  }, [open, t, renderPreview]);
   591→
   592→  return (
   593→    <Dialog open={open} onOpenChange={onOpenChange}>
   594→      <DialogContent aria-describedby={undefined} className="glass-blur max-w-[340px] rounded-3xl p-5">
   595→        <DialogHeader>
   596→          <DialogTitle className="text-sm font-semibold text-white">Live preview</DialogTitle>
   597→        </DialogHeader>
   598→        <div className="relative mx-auto w-full max-w-[260px] overflow-hidden rounded-2xl border border-white/10 bg-black">
   599→          {img ? (
   600→            <img src={img} alt="Styled frame preview" className="w-full" />
   601→          ) : (
   602→            <div className="ac-shimmer relative flex aspect-[9/16] items-center justify-center overflow-hidden bg-white/[0.03]">
   603→              <Loader2 className="h-6 w-6 animate-spin text-violet-300" strokeWidth={1.5} />
   604→            </div>
   605→          )}
   606→          {loading && img && (
   607→            <div className="absolute right-2 top-2 rounded-full bg-black/70 p-1.5" aria-hidden="true">
   608→              <Loader2 className="h-3.5 w-3.5 animate-spin text-violet-300" strokeWidth={1.5} />
   609→            </div>
   610→          )}
   611→        </div>
   612→        <div>
   613→          <div className="mb-2 flex items-center justify-between text-xs text-zinc-400">
   614→            <span>Frame time</span>
   615→            <span className="rounded-full border border-violet-400/30 bg-violet-500/15 px-2.5 py-1 text-[10px] font-bold tabular-nums text-violet-200">
   616→              {t.toFixed(1)}s of {formatDur(job.duration)}
   617→            </span>
   618→          </div>
   619→          <Slider value={[t]} min={0} max={Math.max(1, job.duration - 1)} step={0.5} onValueChange={([v]) => setT(v)} aria-label="Preview frame time" />
   620→        </div>
   621→        <p className="text-[10px] leading-relaxed text-zinc-500">
   622→          Real ffmpeg render of the current style at the chosen moment — exactly what the exported clip looks like.
   623→        </p>
   624→      </DialogContent>
   625→    </Dialog>
   626→  );
   627→}

   629→function TranscriptEditor({
   630→  job, open, onOpenChange, onSaved,
   631→}: { job: Job; open: boolean; onOpenChange: (v: boolean) => void; onSaved: () => void }) {
   632→  const [texts, setTexts] = useState<string[]>([]);
   633→  const [saving, setSaving] = useState(false);
   634→  const { toast } = useToast();
   635→
   636→  useEffect(() => {
   637→    if (open && job.sentences) setTexts(job.sentences.map((s) => s.text));
   638→  }, [open, job.sentences]);
   639→
   640→  const save = async () => {
   641→    setSaving(true);
   642→    try {
   643→      await api.saveTranscript(job.id, texts.map((text) => ({ text })));
   644→      onSaved();
   645→      onOpenChange(false);
   646→    } catch (e) {
   647→      toast({ title: "Save failed", description: String((e as Error).message), variant: "destructive" });
   648→    } finally {
   649→      setSaving(false);
   650→    }
   651→  };
   652→
   653→  return (
   654→    <Dialog open={open} onOpenChange={onOpenChange}>
   655→      <DialogContent aria-describedby={undefined} className="glass-blur max-h-[85vh] max-w-[400px] overflow-hidden rounded-3xl p-5">

   657→          <DialogTitle className="text-sm font-semibold text-white">Transcript editor</DialogTitle>

   659→        <p className="text-[11px] text-zinc-500">
   660→          Fix transcription errors — your edits are burned into the rendered captions.
   661→        </p>
   662→        <div className="max-h-[52vh] space-y-2 overflow-y-auto pr-1">
   663→          {texts.map((text, i) => (
   664→            <div key={i} className="flex gap-2">
   665→              <span className="w-10 shrink-0 pt-2.5 text-right text-[10px] tabular-nums text-zinc-600">
   666→                {Math.floor((job.sentences?.[i]?.start ?? 0) / 60)}:
   667→                {String(Math.round((job.sentences?.[i]?.start ?? 0) % 60)).padStart(2, "0")}
   668→              </span>
   669→              <Textarea
   670→                value={text}
   671→                onChange={(e) => setTexts((arr) => arr.map((v, k) => (k === i ? e.target.value : v)))}
   672→                className="min-h-[44px] resize-none rounded-xl border-white/10 bg-white/[0.04] text-xs text-zinc-100 focus-visible:ring-violet-500/30"
   673→                rows={2}
   674→                aria-label={`Edit sentence ${i + 1}`}
   675→              />
   676→            </div>
   677→          ))}
   678→        </div>
   679→        <button
   680→          type="button"
   681→          onClick={save}
   682→          disabled={saving}
   683→          aria-label="Save transcript and re-render"
   684→          className="ac-cta-ring w-full transition-transform duration-200 active:scale-[0.98] disabled:opacity-60"
   685→        >
   686→          <span className="flex h-12 w-full items-center justify-center gap-2 rounded-full bg-gradient-to-r from-violet-500 to-fuchsia-500 text-sm font-bold text-white">
   687→            {saving ? <Loader2 className="h-4 w-4 animate-spin" strokeWidth={1.5} /> : null}
   688→            Save & re-render with edits
   689→          </span>
   690→        </button>
   691→      </DialogContent>
   692→    </Dialog>
   693→  );
   694→}
